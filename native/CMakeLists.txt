cmake_minimum_required(VERSION 3.16)
project(signlab_native LANGUAGES CXX)
add_library(signlab_native SHARED signlab.cpp)
target_compile_features(signlab_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(signlab_native PRIVATE /W4 /O2 /EHsc)
else()
  target_compile_options(signlab_native PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
